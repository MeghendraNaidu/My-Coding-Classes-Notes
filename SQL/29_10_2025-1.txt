begin
select * from 10kcoders where marks>a;

delimiter $$
create procedure _up(o varchar(100),n varchar(100))
begin
update 10kcoders set name=n where name=o;
end$$
delimiter ;

call _up('ravi','charan');

delimiter $$
create procedure _in(s int,n varchar(100),c varchar(100),m int)
begin
insert into 10kcoders(sno,name,course,marks) values(s,n,c,m);
end$$
delimiter ;


call _in(30,'siva','python',100);

create procedure _del(a int)
